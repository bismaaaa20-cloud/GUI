local function check(name, fn)
	local ok, res = pcall(fn)
	if ok and res == "skip" then
		S += 1
		out("LEWAT  " .. name, "warn")
	elseif ok and res ~= false then
		P += 1
		out("OK     " .. name, "ok")
	else
		F += 1
		out("GAGAL  " .. name .. (ok and "" or (": " .. tostring(res))), "error")
	end
end

--// Pembantu gesture (sentuhan buatan lewat jalur yang sama dengan sentuhan asli)
local function fi(ut, x, y)
	return { UserInputType = Enum.UserInputType[ut or "Touch"], UserInputState = Enum.UserInputState.Begin, KeyCode = Enum.KeyCode.Unknown, Position = Vector3.new(x, y, 0) }
end
local function G(id)
	local found
	for _, g in ipairs(Library._gest or {}) do
		if g.id == id and g.win == win then found = g end
	end
	return found
end
local function move(i, x, y)
	i.Position = Vector3.new(x, y, 0)
	i.UserInputState = Enum.UserInputState.Change
	win._gh.moved(i)
end
local function stop(i)
	i.UserInputState = Enum.UserInputState.End
	win._gh.ended(i)
end
local function origin(g)
	local ap, as = g.hit.AbsolutePosition, g.hit.AbsoluteSize
	local inset = game:GetService("GuiService"):GetGuiInset().Y
	return ap.X + as.X * 0.25, ap.Y + as.Y * 0.5 - inset, as.X * 0.5
end

		assert(ok and n > 0, "impor gagal")
		assert(win.Flags.t_slider == 33, "nilai tidak kembali")
		assert(win:ImportConfig("ngawur") == false, "kode salah harus ditolak")
	end)
	check("P9 simpan, muat, hapus config", function()
		if not fs then return "skip" end
		assert(win:SaveConfig("tes_cfg"))
		assert(win:LoadConfig("tes_cfg"), "muat gagal")
		win:DeleteConfig("tes_cfg")
	end)
	check("P9 profil", function()
		if not fs then return "skip" end
		assert(win:NewProfile("tes_profil"))
		assert(table.find(win:GetProfiles(), "tes_profil"), "profil tidak terdaftar")
		assert(win:DeleteProfile())
		assert(not table.find(win:GetProfiles(), "tes_profil"), "profil tidak terhapus")
	end)
	check("P9 tema kustom", function()
		local prev = Library.ThemeName
		assert(win:SaveTheme("tes_tema"))
		assert(Library.Presets.tes_tema, "preset tidak terdaftar")
		local code = win:ExportTheme("tes_tema")
		assert(code and code:sub(1, 4) == "XT1:", "kode tema salah")
		win:DeleteTheme("tes_tema")
		assert(not Library.Presets.tes_tema, "tema tidak terhapus")
		local ok, nm = win:ImportTheme(code)
		assert(ok, "impor tema gagal")
		win:DeleteTheme(nm)
		if prev then Library:SetTheme(prev) end
	end)
	check("P9 warna dan aksesibilitas", function()
		local c = Library.Theme.Blue
		Library:ApplyColors({ Blue = Color3.fromRGB(255, 0, 0) })
		assert(Library.Theme.Blue == Color3.fromRGB(255, 0, 0), "ApplyColors gagal")
		Library:ApplyColors({ Blue = c })
		win:SetTextScale(115)
		task.wait(0.2)
		win:SetTextScale(100)
		win:SetContrast(true)
		task.wait(0.2)
		win:SetContrast(false)
	end)
	check("P9 cek versi (jaringan)", function()
		if not CFG.TestNetwork then return "skip" end
		win:CheckUpdate(true)
		task.wait(2)
	end)
	check("Kompatibilitas executor", function()
		local c = Library:CheckCompatibility()
		assert(type(c) == "table", "hasil bukan tabel")
		out("Executor: " .. tostring(c.Executor) .. " | tidak didukung: " .. ((c.Unsupported and #c.Unsupported > 0) and table.concat(c.Unsupported, ", ") or "-"))
	end)

	local total = P + F + S
	stats:Set(1, P)
	stats:Set(2, F)
	stats:Set(3, S)
	ringT:Set((total - S) > 0 and P / (total - S) * 100 or 0)
	out(string.format("Selesai %.1f dtk | lulus %d | gagal %d | lewat %d", os.clock() - t0, P, F, S), F == 0 and "ok" or "error")
	win:Notify({ Title = F == 0 and "Semua tes lulus" or (F .. " tes gagal"), Content = "Lihat log di tab Hasil", Type = F == 0 and "success" or "error", Dedupe = false })
	win:SetBanner(nil)
	win:SetGlobalSearch(gs0 ~= false)
	win:_select(hasil)
	running = false
end

win:AddSettings()
win:AddDiagnostics()
